``` r
nyc_clean <- nyc_transit |> 
  rename_with(~ str_replace_all(tolower(.), " ", "_")) |> 
  select(
    line,
    station_name,
    station_latitude,
    station_longitude,
    starts_with("route"), 
    entrance_type,
    entry,
    vending,
    ada
  ) |> 
  mutate(
    
    entry = case_match(
      entry,
      "YES" ~ TRUE,
      "NO" ~ FALSE,
      .default = NA
    ),
 
    across(starts_with("route"), as.character)
  )
```
